Leetcode problem: 518. Coin Change 2

problem link: https://leetcode.com/problems/coin-change-ii/submissions/2023503688

code: 

   class Solution {
public:
    int change(int amount, vector<int>& coins) {
        int n = coins.size();
        vector<vector<long long>> dp(n + 1, vector<long long>(amount + 1, 0));

        for (int i = 0; i < n + 1; i++)
            dp[i][0] = 1;

        for (int i = 1; i < n + 1; i++) {
            for (int j = 1; j < amount + 1; j++) {
                if (coins[i - 1] <= j) {
                    dp[i][j] = dp[i][j - coins[i - 1]] + dp[i - 1][j];
                } else {
                    dp[i][j] = dp[i - 1][j];
                }
                dp[i][j] = min(dp[i][j], (long long)INT_MAX + 1);
            }
        }

        return (int)dp[n][amount];
    }
};
