leetcode problem: 96. Unique Binary Search Trees

link: https://leetcode.com/problems/unique-binary-search-trees/submissions/2025051880

code: -tabulation

class Solution {
public:
    int numTrees(int n) {
        vector<int> dp(n+1, 0);
        dp[0] = dp[1] = 1;

        for(int i = 2; i <= n; i ++) {
            for(int j = 0; j < i; j ++) {
                dp[i] += dp[j] * dp[i - j - 1];
            }
        }
        return dp[n];
    }
};
